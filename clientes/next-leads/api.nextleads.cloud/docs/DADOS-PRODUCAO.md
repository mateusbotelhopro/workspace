## Monitoramento rápido

```bash
# Status por SDR
docker exec nextleads_postgres psql -U postgres -d nextleads -c "
SELECT s.nome AS sdr, COUNT(DISTINCT l.id) AS leads,
  COUNT(m.id) AS mensagens,
  COUNT(m.id) FILTER (WHERE m.\"receivedAt\" >= NOW() - INTERVAL '24 hours') AS ultimas_24h,
  COUNT(w.id) FILTER (WHERE w.status = 'sent') AS alertas_enviados,
  COUNT(w.id) FILTER (WHERE w.status = 'failed') AS falhas,
  COUNT(w.id) FILTER (WHERE w.status LIKE 'sla%') AS alertas_sla,
  MAX(m.\"receivedAt\") AS ultima_mensagem
FROM sdrs s
LEFT JOIN leads l ON l.\"sdrId\" = s.id
LEFT JOIN messages m ON m.\"leadId\" = l.id AND m.direction = 'inbound'
LEFT JOIN whatsapp_logs w ON w.\"sdrId\" = s.id
WHERE s.ativo = true
GROUP BY s.id, s.nome
ORDER BY ultimas_24h DESC, ultima_mensagem DESC NULLS LAST;
"

# Logs em tempo real
docker logs nextleads_api --tail 50 | grep -E "routedSdrId|Alerta enviado|Erro|Unipile sync"
```
